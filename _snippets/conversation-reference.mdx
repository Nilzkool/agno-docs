<!-- @DeepDocs: Updated to match Agent class fields in agent.py -->
<ResponseField name="model" type="Model">
  Model to use for this Agent (e.g., OpenAIChat)
</ResponseField>

<ResponseField name="introduction" type="str">
  Introduction message added to chat history when a run starts
</ResponseField>

<ResponseField name="user_id" type="str">
  Default user identifier for this Agent
</ResponseField>

<ResponseField name="session_id" type="str">
  Session identifier (autogenerated if not set)
</ResponseField>

<ResponseField name="session_name" type="str">
  Human-readable name for this session
</ResponseField>

<ResponseField name="context" type="Dict[str, Any]">
  Arbitrary context available for tools and prompts
</ResponseField>
<ResponseField name="add_context" type="bool">
  If true, include `context` in the user prompt
</ResponseField>
<ResponseField name="resolve_context" type="bool">
  If true, call any functions in `context` before running
</ResponseField>

<ResponseField name="memory" type="Union[AgentMemory, Memory]">
  Memory backend for storing runs and user memories
</ResponseField>
<ResponseField name="enable_agentic_memory" type="bool">
  Allow Agent to manage its own memories
</ResponseField>
<ResponseField name="enable_user_memories" type="bool">
  Create or update user memories after each run
</ResponseField>
<ResponseField name="add_memory_references" type="bool">
  Include memory references in responses when applicable
</ResponseField>

<ResponseField name="add_history_to_messages" type="bool">
  Include recent run history in messages sent to the model
</ResponseField>
<ResponseField name="num_history_runs" type="int">
  Number of previous runs to include in history
</ResponseField>

<ResponseField name="knowledge" type="AgentKnowledge">
  Optional knowledge base for retrieval-augmented generation
</ResponseField>
<ResponseField name="add_references" type="bool">
  Include retrieved references in prompts when knowledge is set
</ResponseField>
<ResponseField name="retriever" type="Callable">
  Custom function to fetch knowledge documents
</ResponseField>
<ResponseField name="references_format" type="Literal['json','yaml']">
  Format for returned knowledge references
</ResponseField>

<ResponseField name="storage" type="Storage">
  Optional storage backend for persisting sessions
</ResponseField>
<ResponseField name="extra_data" type="Dict[str, Any]">
  Arbitrary extra data stored with this Agent
</ResponseField>

<ResponseField name="tools" type="List[Union[Toolkit, Function, Callable]]">
  List of tools (functions) available to the model
</ResponseField>
<ResponseField name="show_tool_calls" type="bool">
  If true, show tool calls in generated responses
</ResponseField>
<ResponseField name="tool_call_limit" type="int">
  Maximum number of tool calls allowed per run
</ResponseField>
<ResponseField name="tool_choice" type="Union[str, Dict]">
  Controls which tool the model may call (e.g., "auto", "none", or specific function)
</ResponseField>

<ResponseField name="system_message" type="Union[str, Callable, Message]">
  Custom system message or function to generate it
</ResponseField>
<ResponseField name="system_message_role" type="str">
  Role assigned to system messages (default: "system")
</ResponseField>
<ResponseField name="create_default_system_message" type="bool">
  If true, build a default system prompt from Agent settings
</ResponseField>

<ResponseField name="user_message" type="Union[str, Callable, Message]">
  Custom user prompt or function to generate it
</ResponseField>
<ResponseField name="user_message_role" type="str">
  Role assigned to user messages (default: "user")
</ResponseField>
<ResponseField name="create_default_user_message" type="bool">
  If true, build a default user prompt based on input message
</ResponseField>

<ResponseField name="response_model" type="Type[BaseModel]">
  Pydantic model for structured outputs
</ResponseField>
<ResponseField name="parse_response" type="bool">
  If true, parse model outputs into `response_model`
</ResponseField>
<ResponseField name="use_json_mode" type="bool">
  Force JSON schema mode for structured outputs
</ResponseField>
<ResponseField name="structured_outputs" type="bool">
  Use native structured outputs if supported by the model
</ResponseField>

<ResponseField name="stream" type="bool">
  Stream response content from the model
</ResponseField>
<ResponseField name="stream_intermediate_steps" type="bool">
  Stream reasoning and tool-call events during a run
</ResponseField>

<ResponseField name="debug_mode" type="bool">
  Enable verbose debug logging
</ResponseField>
<ResponseField name="monitoring" type="bool">
  Send monitoring data to agno.com
</ResponseField>
<ResponseField name="telemetry" type="bool">
  Enable minimal telemetry for analytics
</ResponseField>
